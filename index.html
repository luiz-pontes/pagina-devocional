import streamlit as st # --- INICIALIZAÇÃO DE ESTADOS NA SESSÃO --- if "favoritos" not in st.session_state: st.session_state["favoritos"] = [] if "anotacoes" not in st.session_state: st.session_state["anotacoes"] = {} # --- BARRA LATERAL (SIDEBAR) --- st.sidebar.title("📖 Vida Na Palavra") st.sidebar.text(f"Usuário: {st.session_state.get('user_email', 'flsp1986@hotmail.com')}") if st.sidebar.button("Sair / Logout"): st.session_state.clear() st.rerun() st.sidebar.markdown("---") st.sidebar.subheader("Navegação") # Controle de rotas via Radio Button menu = st.sidebar.radio( "Ir para:", ["Devocional Diário", "Meus Favoritos", "Painel Admin"], label_visibility="collapsed" ) # --- ROTA 1: DEVOCIONAL DIÁRIO --- if menu == "Devocional Diário": st.title("Luz Para o Meu Caminho") # Exemplo de ID único do devocional atual devocional_id = "salmos_119_105" st.subheader("📖 Salmos 119:105") st.write("Lâmpada para os meus pés é tua palavra e luz, para o meu caminho.") # Botão de Favoritar favorito_ativo = devocional_id in st.session_state["favoritos"] btn_label = "⭐ Removido dos Favoritos" if favorito_ativo else "⭐ Adicionar aos Favoritos" if st.button(btn_label): if favorito_ativo: st.session_state["favoritos"].remove(devocional_id) st.success("Removido dos favoritos!") else: st.session_state["favoritos"].append(devocional_id) st.success("Adicionado aos favoritos!") st.rerun() st.markdown("---") st.subheader("💡 Fortalecimento Espiritual do Dia") st.info("Quando a Palavra de Deus guia seus passos, você nunca caminhará na escuridão.") st.markdown("---") st.subheader("📝 Minhas Anotações e Reflexão Pessoal") # Carrega nota salva previamente, se existir nota_atual = st.session_state["anotacoes"].get(devocional_id, "") texto_nota = st.text_area( "Escreva o que Deus falou ao seu coração hoje:", value=nota_atual, placeholder="Digite aqui sua oração, insight ou reflexão espiritual...", height=150 ) if st.button("💾 Salvar Anotação"): st.session_state["anotacoes"][devocional_id] = texto_nota # Se você usa banco de dados SQLite, chame sua função de salvar aqui: # salvar_anotacao_db(user_id, devocional_id, texto_nota) st.success("Anotação salva com sucesso!") # --- ROTA 2: MEUS FAVORITOS E ANOTAÇÕES SALVAS --- elif menu == "Meus Favoritos": st.title("⭐ Meus Favoritos e Anotações") if not st.session_state["favoritos"] and not st.session_state["anotacoes"]: st.warning("Você ainda não salvou nenhum devocional ou anotação.") else: tab1, tab2 = st.tabs(["Devocionais Favoritos", "Minhas Reflexões Salvas"]) with tab1: if st.session_state["favoritos"]: for item in st.session_state["favoritos"]: with st.expander(f"📌 Devocional: {item}"): st.write("Conteúdo salvo do devocional...") else: st.write("Nenhum devocional favoritado ainda.") with tab2: if st.session_state["anotacoes"]: for key, val in st.session_state["anotacoes"].items(): if val.strip(): st.markdown(f"**Reflexão para `{key}`:**") st.info(val) else: st.write("Nenhuma reflexão anotada ainda.") # --- ROTA 3: PAINEL ADMIN --- elif menu == "Painel Admin": st.title("⚙️ Painel de Administração") st.write("Gerenciamento de conteúdos do aplicativo.")
